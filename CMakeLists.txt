cmake_minimum_required(VERSION 3.8)
project(teambot_bringup)
find_package(ament_cmake REQUIRED)
find_package(Threads REQUIRED)
add_library(software_egl SHARED egl/software_egl.c)
target_link_libraries(software_egl PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
target_compile_options(software_egl PRIVATE -Wall -Wextra)
install(TARGETS software_egl LIBRARY DESTINATION lib)
install(DIRECTORY launch worlds config rviz urdf DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(PROGRAMS scripts/check_sim.py DESTINATION lib/${PROJECT_NAME})
ament_package()
